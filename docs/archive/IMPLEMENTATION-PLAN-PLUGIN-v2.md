**版本**: v2.4.0-L0L1L2

**更新日期**: 2026-03-24

## 一、架构目标

### 1.1 核心原则

- **强制分层**: abstract 和 overview 为**必填参数**，调用方必须提供
- **OpenCode 生成**: 由 OpenCode 主代理或子代理在调用前生成摘要
- **无自动生成**: 插件端**不调用任何 LLM** 生成内容
- **验证严格**: 长度超出建议时警告（不拒绝），格式错误报错

### 1.2 存储结构

```
~/.opencode/memory/
├── MEMORY.md                    # 全局轻量索引（最近20条L0）
├── link-map.json                # 全局关联映射
├── SOUL.md, AGENTS.md, etc.    # 核心记忆文件
└── timeline/
    └── 2026/
        └── 03/
            └── 23/
                ├── entry-001.md      # 完整三层内容（L0/L1/L2）
                ├── entry-002.md      # 完整三层内容
                ├── .overview.md      # 日概览（当天所有L0摘要）
                └── .index.json       # 快速索引

~/.opencode/memory/.archive/     # 本地归档（压缩）
    └── 2026-01-archive.zip      # 按月归档
```

```javascript
memory_write: tool({
  description: 'Write an entry to long-term memory with L0/L1/L2 layered structure. abstract and overview are REQUIRED and must be generated by the calling agent before invocation.',
  args: {
    content: tool.schema.string().describe('L2: The complete content (required)'),

    // 改为必填！
    abstract: tool.schema
      .string()
      .describe('L0: One-sentence summary (recommended ≤100 chars). REQUIRED - must be generated by caller'),

    // 改为必填！
    overview: tool.schema
      .string()
      .describe('L1: Key points summary (recommended ≤500 chars). REQUIRED - must be generated by caller'),

    type: tool.schema.string().optional().default('general'),
    tags: tool.schema.array(tool.schema.string()).optional().default([]),

    // 移除 auto_generate 参数
  },
  async execute(args) {
    try {
      const { content, abstract, overview, type, tags } = args;

      // 严格验证：abstract 和 overview 必须存在且非空
      if (!abstract || abstract.trim().length === 0) {
        return `❌ Error: abstract is required. Generate it before calling memory_write.`;
      }

      if (!overview || overview.trim().length === 0) {
        return `❌ Error: overview is required. Generate it before calling memory_write.`;
      }

      // 验证长度并截断
      let finalAbstract = abstract.trim();
      let finalOverview = overview.trim();

      if (finalAbstract.length > 100) {
        finalAbstract = finalAbstract.substring(0, 97) + '...';
        console.warn(`[memory_write] Abstract truncated from ${abstract.length} to 100 chars`);
      }

      if (finalOverview.length > 500) {
        finalOverview = finalOverview.substring(0, 497) + '...';
        console.warn(`[memory_write] Overview truncated from ${overview.length} to 500 chars`);
      }

      const layers = {
        content: content.trim(),
        abstract: finalAbstract,
        overview: finalOverview
      };

      const metadata = {
        type,
        tags,
        projectId: await resolveProjectId(),
      };

      // 1. 写入 timeline
      const result = await writeEntryToTimeline(layers, metadata);

      // 2. 更新索引文件
      await Promise.all([
        updateDayOverview(result.dayDir, { ...result.entry, fileName: result.fileName }),
        updateMemoryIndex(result.entry, result.filePath),
        updateLinkMap(result.entry, result.filePath),
      ]);

      // 3. 异步生成日索引
      generateDayIndex(result.dayDir).catch(e =>
        console.warn('[Index] Failed:', e.message)
      );

      // 4. 后端同步
      const syncResult = await syncToBackend(result.entry, layers);

      return `✅ Memory saved\n- Abstract: ${finalAbstract.substring(0, 50)}${finalAbstract.length > 50 ? '...' : ''}\n- File: ${result.filePath.replace(MEMORY_DIR, '')}\n- Backend: ${syncResult.status}`;

    } catch (e) {
      return `❌ Error: ${e.message}`;
    }
  },
}),
```

async function writeEntryToTimeline(layers, metadata) {
  const now = new Date();
  const [year, month, day] = [
    now.getFullYear(),
    String(now.getMonth() + 1).padStart(2, "0"),
    String(now.getDate()).padStart(2, "0"),
  ];

  const dayDir = path.join(TIMELINE_DIR, String(year), month, day);

  if (!fs.existsSync(dayDir)) {
    fs.mkdirSync(dayDir, { recursive: true });
  }

  const fileName = `entry-${Date.now()}-${Math.random().toString(36).substring(2, 5)}.md`;
  const filePath = path.join(dayDir, fileName);

  const entryId = `mem_${year}${month}${day}_${Math.random().toString(36).substring(2, 6)}`;

  // 构建内容 - 严格三层格式
  const content = `---
id: ${entryId}
date: ${now.toISOString()}
type: ${metadata.type || "general"}
tags: [${(metadata.tags || []).join(", ")}]
project: ${metadata.projectId || "global"}
memory_id: pending
---

# Abstract
${layers.abstract}

## Overview
${layers.overview}

## Content
${layers.content}

---
`;

  fs.writeFileSync(filePath, content, "utf-8");

  return {
    filePath,
    fileName,
    entryId,
    dayDir,
    entry: {
      id: entryId,
      abstract: layers.abstract,
      type: metadata.type,
      tags: metadata.tags || [],
    },
  };
}

async function updateDayOverview(dayDir, entry) {
  const overviewPath = path.join(dayDir, ".overview.md");
  const dateStr = path.basename(dayDir);

  let lines = ["# " + dateStr + " 记忆概览\n"];

  if (fs.existsSync(overviewPath)) {
    lines = fs.readFileSync(overviewPath, "utf-8").split("\n");
  }

  // 添加新条目
  const shortAbstract =
    entry.abstract.length > 80
      ? entry.abstract.substring(0, 77) + "..."
      : entry.abstract;

  lines.splice(2, 0, `- [${entry.type}] ${shortAbstract} → ${entry.fileName}`);

  // 限制100条
  if (lines.length > 102) {
    lines.splice(102);
  }

  // 更新计数
  const count = lines.filter((l) => l.startsWith("- [")).length;
  lines[0] = `# ${dateStr} 记忆概览（${count} entries）`;

  fs.writeFileSync(overviewPath, lines.join("\n"), "utf-8");
}

#### 任务 2.2: MEMORY.md 更新（限制20条）

```javascript
async function updateMemoryIndex(entry, filePath) {
  const maxEntries = 20;

  let lines = [
    "# 长期记忆索引 v2.3",
    `**最后更新**: ${new Date().toISOString().split("T")[0]}`,
    "## 最近记忆\n",
  ];

  if (fs.existsSync(MEMORY_FILE)) {
    const content = fs.readFileSync(MEMORY_FILE, "utf-8");
    lines = content.split("\n");
  }

  const relativePath = filePath
    .replace(MEMORY_DIR + path.sep, "")
    .replace(/\\/g, "/");
  const shortAbstract =
    entry.abstract.length > 60
      ? entry.abstract.substring(0, 57) + "..."
      : entry.abstract;

  const entryLine = `- [${entry.type}] ${shortAbstract} → ${relativePath}`;

  // 插入到"## 最近记忆"后
  const insertIdx = lines.findIndex((l) => l.startsWith("## ")) + 1;
  lines.splice(insertIdx, 0, entryLine);

  // 只保留20条
  const entryLines = lines.filter((l) => l.startsWith("- ["));
  if (entryLines.length > maxEntries) {
    let count = 0;
    for (let i = insertIdx; i < lines.length; i++) {
      if (lines[i].startsWith("- [")) {
        count++;
        if (count > maxEntries) {
          lines.splice(i, 1);
          i--;
        }
      }
    }
  }

  fs.writeFileSync(MEMORY_FILE, lines.join("\n"), "utf-8");
}
```

async function updateLinkMap(entry, filePath) {
  let linkMap = { version: "2.3.0", entries: {} };

  if (fs.existsSync(LINK_MAP_FILE)) {
    linkMap = JSON.parse(fs.readFileSync(LINK_MAP_FILE, "utf-8"));
  }

  const relativePath = filePath
    .replace(MEMORY_DIR + path.sep, "")
    .replace(/\\/g, "/");

  linkMap.entries[entry.id] = {
    path: relativePath,
    abstract: entry.abstract,
    type: entry.type,
    tags: entry.tags,
  };

  linkMap.updated_at = new Date().toISOString();

  fs.writeFileSync(LINK_MAP_FILE, JSON.stringify(linkMap, null, 2), "utf-8");
}

### Phase 3: memory_read 改造（1.5小时）

```javascript
memory_read: tool({
  description: 'Read memory entry by ID with level support',
  args: {
    entry_id: tool.schema.string().describe('Entry ID (required)'),
    level: tool.schema.number().optional().default(2)
      .describe('0=abstract, 1=abstract+overview, 2=full'),
  },
  async execute(args) {
    const { entry_id, level } = args;

    if (!fs.existsSync(LINK_MAP_FILE)) {
      return `❌ Link map not found`;
    }

    const linkMap = JSON.parse(fs.readFileSync(LINK_MAP_FILE, 'utf-8'));
    const entry = linkMap.entries[entry_id];

    if (!entry) {
      return `❌ Entry not found: ${entry_id}`;
    }

    const filePath = path.join(MEMORY_DIR, entry.path);
    if (!fs.existsSync(filePath)) {
      return `❌ File not found: ${entry.path}`;
    }

    const content = fs.readFileSync(filePath, 'utf-8');

    if (level === 0) {
      const match = content.match(/# Abstract\n([\s\S]*?)(?=\n## |\n---|$)/);
      return match ? match[1].trim() : 'No abstract found';
    }

    if (level === 1) {
      const abstract = content.match(/# Abstract\n([\s\S]*?)(?=\n## |$)/)?.[1]?.trim() || '';
      const overview = content.match(/## Overview\n([\s\S]*?)(?=\n## |$)/)?.[1]?.trim() || '';
      return `# Abstract\n${abstract}\n\n## Overview\n${overview}`;
    }

    return content;
  },
}),
```
