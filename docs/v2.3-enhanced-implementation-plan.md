### P3: 冲突解决框架 - 自动裁决 (2小时) ⭐

**目标**: 智能检测和自动解决冲突

**详细步骤**:

1. **冲突检测** (30分钟)
```javascript
function detectConflict(localEntry, backendEntry) {
  // 双方都修改过
  const localModified = localEntry.updated_at;
  const backendModified = backendEntry.updated_at;
  
  if (localModified && backendModified) {
    // 计算时间差
    const timeDiff = Math.abs(localModified - backendModified);
    
    // 相似度检查
    const similarity = cosineSimilarity(
      localEntry.embedding,
      backendEntry.embedding
    );
    
    return {
      type: timeDiff < 3600000 ? 'simultaneous' : 'sequential',
      similarity,
      localNewer: localModified > backendModified
    };
  }
  
  return null;
}
```

2. **自动裁决策略** (1小时)
```javascript
function autoResolve(conflict) {
  // 策略1: 时间戳裁决（简单场景）
  if (conflict.type === 'sequential') {
    return conflict.localNewer ? 'USE_LOCAL' : 'USE_BACKEND';
  }
  
  // 策略2: 内容质量评估
  const localQuality = assessQuality(conflict.local);
  const backendQuality = assessQuality(conflict.backend);
  
  if (Math.abs(localQuality - backendQuality) > 0.3) {
    return localQuality > backendQuality ? 'USE_LOCAL' : 'USE_BACKEND';
  }
  
  // 策略3: 相似度合并（高相似）
  if (conflict.similarity > 0.9) {
    return 'MERGE';
  }
  
  // 需要人工决策
  return 'MANUAL';
}
```

3. **合并算法** (30分钟)
```javascript
function mergeEntries(local, backend) {
  // 合并元数据
  const merged = {
    ...backend,  // 以backend为基准
    content: local.content.length > backend.content.length 
      ? local.content  // 选更详细的
      : backend.content,
    tags: [...new Set([...local.tags, ...backend.tags])],  // 合并标签
    updated_at: new Date(),
    merge_history: [
      ...(backend.merge_history || []),
      { from: 'local', at: local.updated_at }
    ]
  };
  
  return merged;
}
```

```javascript
// memory_resolve_conflict 工具
async function resolveConflict({ conflict_id, resolution }) {
  // resolution: 'USE_LOCAL' | 'USE_BACKEND' | 'MERGE'
  
  const conflict = await loadConflict(conflict_id);
  
  switch (resolution) {
    case 'USE_LOCAL':
      await backendUpdate(conflict.local);
      break;
    case 'USE_BACKEND':
      await localUpdate(conflict.backend);
      break;
    case 'MERGE':
      const merged = mergeEntries(conflict.local, conflict.backend);
      await backendUpdate(merged);
      await localUpdate(merged);
      break;
  }
  
  await markResolved(conflict_id);
}
```

```javascript
async function batchResolve({ strategy }) {

      ? conflict.recommendation
```
